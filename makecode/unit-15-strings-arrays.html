15.1 字串基礎

15.2 字串串接與顯示

使用 + 運算符組合字串與變數：

// 顯示倒數訊息
let count = 10
basic.forever(function () {
    basic.showString("T-" + count)
    count -= 1
    if (count < 0) {
        basic.showString("GO!")
        count = 10
    }
    basic.pause(1000)
})

格式化數字

// 顯示溫度訊息
basic.forever(function () {
    let temp = input.temperature()
    basic.showString("Temp:" + temp + "C")
    basic.pause(2000)
})
重點：在 MakeCode 的積木模式中，使用「顯示字串」積木搭配 [A] + [B] 積木來串接文字。自動轉換的 JavaScript 程式碼會使用 + 運算符。

15.4 陣列操作

遍歷陣列

// 顯示陣列中的每個元素
let fruits = ["Apple", "Banana", "Cherry"]
for (let i = 0; i < fruits.length; i++) {
    basic.showString(fruits[i])
    basic.pause(1000)
}

搜尋陣列

// 在陣列中搜尋特定值
let numbers = [3, 7, 2, 9, 5]
let target = 7
let found = false
for (let i = 0; i < numbers.length; i++) {
    if (numbers[i] === target) {
        basic.showString("Found at " + i)
        found = true
        break
    }
}
if (!found) {
    basic.showString("Not found")
}

15.6 實作練習：隨機問答遊戲

  1. 建立題目陣列和答案陣列
  2. 隨機選一題顯示
  3. 按 A 表示「是」，按 B 表示「否」
  4. 判斷對錯並計分
let questions = [
    "地球是圓的?",
    "太陽從東方升起?",
    "1+1=3?",
    "水的沸點是100度?"
]
let answers = [true, true, false, true]
let score = 0
let qIndex = 0

basic.showString(questions[qIndex])

input.onButtonPressed(Button.A, function () {
    if (answers[qIndex] === true) {
        score += 1
        basic.showIcon(IconNames.Yes)
    } else {
        basic.showIcon(IconNames.No)
    }
    nextQuestion()
})

input.onButtonPressed(Button.B, function () {
    if (answers[qIndex] === false) {
        score += 1
        basic.showIcon(IconNames.Yes)
    } else {
        basic.showIcon(IconNames.No)
    }
    nextQuestion()
})

function nextQuestion() {
    basic.pause(1000)
    qIndex = (qIndex + 1) % questions.length
    basic.showString(questions[qIndex])
}
